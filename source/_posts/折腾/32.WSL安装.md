## 内存精简

### 第一步：诊断脚本

先运行这个，生成当前状态报告到桌面 `win_report.txt`。每次优化前后都可跑一次做对比。

```powershell
$out = "$env:USERPROFILE\Desktop\win_report.txt"
Remove-Item $out -ErrorAction SilentlyContinue
 
"###### 1. 内存概况 ######`n" | Out-File $out -Append
Get-CimInstance Win32_OperatingSystem |
  Select-Object @{n='总内存GB';e={[math]::Round($_.TotalVisibleMemorySize/1MB,2)}},
                @{n='可用GB';e={[math]::Round($_.FreePhysicalMemory/1MB,2)}} |
  Format-List | Out-File $out -Append
 
"`n###### 2. 正在运行的服务(按内存排序) ######`n" | Out-File $out -Append
$procs = Get-Process -ErrorAction SilentlyContinue
Get-CimInstance Win32_Service | Where-Object State -eq 'Running' | ForEach-Object {
  $p = $procs | Where-Object Id -eq $_.ProcessId | Select-Object -First 1
  [PSCustomObject]@{
    服务名   = $_.Name
    显示名   = $_.DisplayName
    启动类型 = $_.StartMode
    PID      = $_.ProcessId
    内存MB   = if($p){[math]::Round($p.WorkingSet64/1MB,1)}else{0}
  }
} | Sort-Object 内存MB -Descending |
  Format-Table -AutoSize | Out-File $out -Append -Width 220
 
"`n###### 3. 设为自动启动但当前未运行的服务 ######`n" | Out-File $out -Append
Get-CimInstance Win32_Service | Where-Object {$_.StartMode -eq 'Auto' -and $_.State -ne 'Running'} |
  Select-Object Name, DisplayName, State |
  Format-Table -AutoSize | Out-File $out -Append -Width 220
 
"`n###### 4. 占用内存最高的进程 Top 30 ######`n" | Out-File $out -Append
Get-Process | Sort-Object WorkingSet64 -Descending | Select-Object -First 30 `
  @{n='进程';e={$_.ProcessName}}, Id, @{n='内存MB';e={[math]::Round($_.WorkingSet64/1MB,1)}} |
  Format-Table -AutoSize | Out-File $out -Append -Width 220
 
"`n###### 5. 启动项(注册表 Run) ######`n" | Out-File $out -Append
$runKeys = @(
 'HKLM:\Software\Microsoft\Windows\CurrentVersion\Run',
 'HKLM:\Software\Microsoft\Windows\CurrentVersion\RunOnce',
 'HKCU:\Software\Microsoft\Windows\CurrentVersion\Run',
 'HKCU:\Software\Microsoft\Windows\CurrentVersion\RunOnce'
)
foreach($k in $runKeys){
  if(Test-Path $k){
    "[$k]" | Out-File $out -Append
    (Get-ItemProperty $k).PSObject.Properties | Where-Object {$_.Name -notlike 'PS*'} |
      Select-Object Name, Value | Format-Table -AutoSize | Out-File $out -Append -Width 220
  }
}
 
"`n###### 6. 已启用的计划任务 ######`n" | Out-File $out -Append
Get-ScheduledTask | Where-Object {$_.State -ne 'Disabled'} |
  Select-Object TaskPath, TaskName, State |
  Sort-Object TaskPath | Format-Table -AutoSize | Out-File $out -Append -Width 220
 
"`n###### 7. Defender 状态 ######`n" | Out-File $out -Append
Get-MpComputerStatus -ErrorAction SilentlyContinue |
  Select-Object AMRunningMode, RealTimeProtectionEnabled, AntivirusEnabled |
  Format-List | Out-File $out -Append
 
"`n###### 8. 系统版本 ######`n" | Out-File $out -Append
Get-CimInstance Win32_OperatingSystem | Select-Object Caption, Version, BuildNumber, OSArchitecture |
  Format-List | Out-File $out -Append
 
Start-Process notepad $out
```

$stamp  = Get-Date -Format 'yyyyMMdd_HHmmss'
$backup = "$env:USERPROFILE\Desktop\svc_backup_$stamp.csv"

    Export-Csv $backup -NoTypeInformation -Encoding UTF8
Write-Host "[备份] $backup" -ForegroundColor Cyan
 
# ---- 辅助函数：禁用服务，失败则回退到注册表 Start=4 ----
function Disable-Svc($name) {
    try {
        Stop-Service $name -Force -ErrorAction SilentlyContinue
        Set-Service  $name -StartupType Disabled -ErrorAction Stop
        Write-Host "[OK ] 已禁用 $name" -ForegroundColor Green
    } catch {
        $p = "HKLM:\SYSTEM\CurrentControlSet\Services\$name"
        if (Test-Path $p) {
            Set-ItemProperty $p -Name Start -Value 4 -ErrorAction SilentlyContinue
            Write-Host "[REG] $name 已写 Start=4（重启生效）" -ForegroundColor Yellow
        } else {
            Write-Host "[跳过] $name : $($_.Exception.Message)" -ForegroundColor DarkYellow
        }
    }
}
 
# ==== A 档：🟢 直接禁用（遥测/便利功能，无系统风险）====
$tierA = @(
    'DiagTrack'            # 连接用户体验与遥测（约 66M）
    'wuqisvc'             # 使用与质量见解（遥测）
    'InventorySvc'        # 清单与兼容性评估（遥测）

    'WSAIFabricSvc'       # AI Fabric（摄像头特效相关）
    'webthreatdefsvc'     # Web 威胁防护（SmartScreen）
    'WpnService'          # 推送通知系统服务（禁后无 toast 通知）
    'WpnUserService'      # 推送通知用户服务（模板）
    'CDPSvc'              # 连接设备平台（跨设备/附近共享）
    'CDPUserSvc'          # 连接设备平台用户服务（模板）
    'webthreatdefusersvc' # Web 威胁防护用户服务（模板）
)
$tierA | ForEach-Object { Disable-Svc $_ }
 
# ==== 启动项清理 ====
# Defender 托盘图标
Remove-ItemProperty 'HKLM:\Software\Microsoft\Windows\CurrentVersion\Run' -Name SecurityHealth -ErrorAction SilentlyContinue
# Edge 静默自启（注意：GUID 因机器而异，需先跑诊断脚本第 5 节确认实际名称）
# Remove-ItemProperty 'HKCU:\Software\Microsoft\Windows\CurrentVersion\Run' -Name 'MicrosoftEdgeAutoLaunch_XXXX' -ErrorAction SilentlyContinue
 
# ==== 遥测/更新类计划任务 ====
$tasks = @(
    '\Microsoft\Windows\Application Experience\Microsoft Compatibility Appraiser Exp'
    '\Microsoft\Windows\Application Experience\PcaPatchDbTask'
    '\Microsoft\Windows\Application Experience\StartupAppTask'
    '\Microsoft\Windows\Customer Experience Improvement Program\Consolidator'
    '\Microsoft\Windows\Customer Experience Improvement Program\UsbCeip'
    '\Microsoft\Windows\Feedback\Siuf\DmClient'
    '\Microsoft\Windows\Feedback\Siuf\DmClientOnScenarioDownload'
    '\Microsoft\Windows\Windows Error Reporting\QueueReporting'
    '\Microsoft\Windows\DiskDiagnostic\Microsoft-Windows-DiskDiagnosticDataCollector'
)
foreach ($t in $tasks) {
    $tp = Split-Path $t; $tn = Split-Path $t -Leaf
    Disable-ScheduledTask -TaskPath "$tp\" -TaskName $tn -ErrorAction SilentlyContinue | Out-Null
}
 
# Edge 更新计划任务（用通配符，名字带 {} 花括号）
Get-ScheduledTask -TaskName 'MicrosoftEdgeUpdateTask*' -ErrorAction SilentlyContinue | Disable-ScheduledTask

### 第六步：搜索界面瘦身

**目标**：让开始菜单搜索（`SearchHost`）不再联网，甩掉它拉起的 `msedgewebview2` 进程（省一两百 MB）。

```powershell
# 方法一：搜索的 Web/云开关（机器级）
$k = 'HKLM:\SOFTWARE\Policies\Microsoft\Windows\Windows Search'
New-Item $k -Force | Out-Null
Set-ItemProperty $k -Name ConnectedSearchUseWeb -Value 0 -Type DWord
Set-ItemProperty $k -Name AllowCloudSearch      -Value 0 -Type DWord
Set-ItemProperty $k -Name DisableWebSearch      -Value 1 -Type DWord
 
# 方法二：关闭搜索框网页建议（用户级，更直接）
$s = 'HKCU:\SOFTWARE\Policies\Microsoft\Windows\Explorer'
New-Item $s -Force | Out-Null
Set-ItemProperty $s -Name DisableSearchBoxSuggestions -Value 1 -Type DWord
```

> 🔑 **关键**：写完后必须 **注销当前用户并重新登录**（不是重启！），策略才对会话生效。登录后看任务管理器，`SearchHost` 下的
> `msedgewebview2` 应消失。

### 第十步：注销用户GUI

```bat
@echo off
setlocal enabledelayedexpansion
title Session Logoff Tool

REM ============================================================
REM  Session Logoff Tool
REM  Lists sessions via qwinsta, logs off the chosen one.
REM  Logs off the WHOLE session (desktop + all user programs).
REM  Must run as Administrator to log off other users.
REM ============================================================

REM --- Admin check ---
net session >nul 2>&1
if %errorlevel% neq 0 (
    echo [ERROR] Please run as Administrator.
    echo         Right-click this file -^> Run as administrator
    echo.
    pause
    exit /b 1
)

:MENU
cls
echo ============================================================
echo   Current Sessions
echo ============================================================
echo.

set idx=0
for /f "delims=" %%L in ('set MAP_ 2^>nul') do set "%%L="

REM --- Parse qwinsta, skip header line ---
for /f "skip=1 tokens=1,2,3,4" %%a in ('qwinsta 2^>nul') do (
    set "c1=%%a"
    set "c2=%%b"
    set "c3=%%c"
    set "c4=%%d"
    set "c1=!c1:>=!"

    REM Decide layout: if c2 is a number -^> no username column
    echo !c2!| findstr /r "^[0-9][0-9]*$" >nul
    if !errorlevel! equ 0 (
        set "uname=(none)"
        set "sid=!c2!"
        set "sstate=!c3!"
    ) else (
        set "uname=!c2!"
        set "sid=!c3!"
        set "sstate=!c4!"
    )

    REM Only list rows whose sid is numeric
    echo !sid!| findstr /r "^[0-9][0-9]*$" >nul
    if !errorlevel! equ 0 (
        set /a idx+=1
        set "MAP_!idx!=!sid!"
        set "USR_!idx!=!uname!"
        echo   [!idx!]  Session:!c1!  User:!uname!  ID:!sid!  State:!sstate!
    )
)

echo.
echo ============================================================
if %idx% equ 0 (
    echo   No sessions parsed. Run qwinsta manually to check.
    echo.
    pause
    exit /b 1
)
echo   Enter a number to log off, or Q to quit.
echo ============================================================
echo.

set "choice="
set /p "choice=Select session number: "

if /i "%choice%"=="Q" exit /b 0

echo %choice%| findstr /r "^[0-9][0-9]*$" >nul
if %errorlevel% neq 0 (
    echo.
    echo [ERROR] Please enter a valid number.
    timeout /t 2 >nul
    goto MENU
)

set "sel=!MAP_%choice%!"
set "seluser=!USR_%choice%!"

if "!sel!"=="" (
    echo.
    echo [ERROR] Number %choice% not found.
    timeout /t 2 >nul
    goto MENU
)

echo.
echo ------------------------------------------------------------
echo   About to log off:  ID=!sel!   User=!seluser!
echo   WARNING: all programs in this session will be terminated!
echo ------------------------------------------------------------
set "confirm="
set /p "confirm=Confirm logoff? (Y/N): "
if /i not "!confirm!"=="Y" (
    echo Cancelled.
    timeout /t 2 >nul
    goto MENU
)

echo.
echo Logging off session !sel! ...
logoff !sel! >nul 2>&1
if !errorlevel! equ 0 (
    echo [OK] Session !sel! logged off.
) else (
    echo [FAIL] Could not log off session !sel! ^(code !errorlevel!^).
    echo        Possibly insufficient rights or a protected/current session.
)

echo.
pause
goto MENU
```
